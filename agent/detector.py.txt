"""Multi-signal detection: canary triggers, mass events, entropy spikes."""
import math
import time
from collections import Counter, deque
from pathlib import Path
from . import config

def shannon_entropy(data: bytes) -> float:
    if not data: return 0.0
    counts = Counter(data)
    length = len(data)
    return -sum((c / length) * math.log2(c / length) for c in counts.values())

def file_entropy(path: Path, max_bytes=None) -> float:
    max_bytes = max_bytes or config.ENTROPY_MAX_BYTES
    try:
        with open(path, "rb") as f: return shannon_entropy(f.read(max_bytes))
    except Exception: return 0.0

class Detector:
    def __init__(self):
        self._events = deque(maxlen=4000)
        self._entropy_baseline = {}
        self.suppressed_paths = set()

    def suppress(self, path): self.suppressed_paths.add(str(path))
    def unsuppress(self, path): self.suppressed_paths.discard(str(path))

    def record_event(self, path, event_type, entropy=None):
        path = str(path)
        if path in self.suppressed_paths: return None
        now = time.time()
        self._events.append((now, path, event_type))
        signals, score = [], 0

        if Path(path).name.startswith(config.CANARY_PREFIX) and event_type != "created":
            score += 100
            signals.append("canary file modified")

        if entropy is not None:
            prev = self._entropy_baseline.get(path)
            self._entropy_baseline[path] = entropy
            if (prev is not None and entropy > config.ENTROPY_THRESHOLD and (entropy - prev) > config.ENTROPY_DELTA):
                score += 30
                signals.append(f"entropy spike {prev:.2f} -> {entropy:.2f}")

        if Path(path).suffix.lower() in config.SUSPICIOUS_EXTENSIONS:
            score += 25
            signals.append("suspicious extension")

        recent_mods = [e for e in self._events if e[2] == "modified" and now - e[0] <= config.MASS_MOD_WINDOW]
        if len(recent_mods) >= config.MASS_MOD_THRESHOLD:
            score += 50
            signals.append(f"{len(recent_mods)} modifications in {config.MASS_MOD_WINDOW}s")

        recent_renames = [e for e in self._events if e[2] == "moved" and now - e[0] <= config.MASS_RENAME_WINDOW]
        if len(recent_renames) >= config.MASS_RENAME_THRESHOLD:
            score += 40
            signals.append(f"{len(recent_renames)} renames in {config.MASS_RENAME_WINDOW}s")

        if score == 0: return None
        return {"score": score, "severity": self._severity(score), "signals": signals, "path": path}

    @staticmethod
    def _severity(score):
        if score >= config.RISK_CRITICAL: return "critical"
        if score >= config.RISK_HIGH: return "high"
        if score >= config.RISK_MEDIUM: return "medium"
        return "low"

    def recent_affected_paths(self, window=30):
        now = time.time()
        return {e[1] for e in self._events if now - e[0] <= window}

    def recent_events(self, limit=100):
        out = []
        for ts, path, event_type in list(self._events)[-limit:]:
            out.append({"timestamp": ts, "path": path, "event": event_type})
        return out

    def threat_level(self, window=10):
        now = time.time()
        recent = sum(1 for e in self._events if now - e[0] <= window)
        level = min(100, recent * 4)
        if level >= 70: label = "critical"
        elif level >= 40: label = "high"
        elif level >= 15: label = "medium"
        else: label = "safe"
        return {"level": level, "label": label, "recent_events": recent, "window": window}
