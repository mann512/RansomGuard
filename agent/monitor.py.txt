"""Watchdog-based file system monitor with skip filter."""
import time
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
from . import config
from .detector import Detector, file_entropy

def _should_skip(path) -> bool:
    try: parts = Path(path).parts
    except Exception: return False
    for skip in config.MONITOR_SKIP_DIRS:
        if skip in parts: return True
    return False

class Handler(FileSystemEventHandler):
    def __init__(self, detector: Detector, on_signal):
        super().__init__()
        self.detector = detector
        self.on_signal = on_signal
        self._last = {}

    def _debounce(self, key, seconds=0.4):
        now = time.time()
        if key in self._last and now - self._last[key] < seconds: return False
        self._last[key] = now
        return True

    def on_modified(self, event):
        if event.is_directory or _should_skip(event.src_path): return
        path = Path(event.src_path)
        if not self._debounce(("mod", str(path))): return
        sig = self.detector.record_event(path, "modified", entropy=file_entropy(path))
        if sig: self.on_signal(sig)

    def on_created(self, event):
        if event.is_directory or _should_skip(event.src_path): return
        sig = self.detector.record_event(Path(event.src_path), "created")
        if sig: self.on_signal(sig)

    def on_deleted(self, event):
        if event.is_directory or _should_skip(event.src_path): return
        sig = self.detector.record_event(Path(event.src_path), "deleted")
        if sig: self.on_signal(sig)

    def on_moved(self, event):
        if event.is_directory or _should_skip(event.dest_path): return
        sig = self.detector.record_event(Path(event.dest_path), "moved")
        if sig: self.on_signal(sig)

class FileMonitor:
    def __init__(self, folders, on_signal):
        self.folders = [str(Path(f)) for f in folders]
        self.on_signal = on_signal
        self.detector = Detector()
        self.observer = None

    def start(self):
        handler = Handler(self.detector, self.on_signal)
        self.observer = Observer()
        for folder in self.folders:
            if Path(folder).exists(): self.observer.schedule(handler, folder, recursive=True)
        self.observer.start()

    def stop(self):
        if self.observer:
            self.observer.stop()
            self.observer.join(timeout=5)
            self.observer = None
