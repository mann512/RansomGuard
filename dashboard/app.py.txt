"""Flask dashboard + JSON API for RansomGuard."""
from flask import Flask, jsonify, request, render_template
from agent import canary
from agent.core import RansomGuard

app = Flask(__name__, template_folder="templates", static_folder="static")
core = RansomGuard()

@app.get("/api/status")
def api_status(): return jsonify(core.get_status())

@app.get("/api/alerts")
def api_alerts(): return jsonify(core.db.list_alerts(limit=int(request.args.get("limit", 50))))

@app.get("/api/alerts/<int:alert_id>")
def api_alert(alert_id):
    a = core.db.get_alert(alert_id)
    if not a: return jsonify({"error": "not found"}), 404
    a["backups"] = core.db.backups_for_alert(alert_id)
    return jsonify(a)

@app.post("/api/alerts/<int:alert_id>/restore")
def api_restore(alert_id): return jsonify(core.restore_alert(alert_id))

@app.get("/api/activity")
def api_activity(): return jsonify(core.get_activity(limit=int(request.args.get("limit", 100))))

@app.get("/api/threat")
def api_threat(): return jsonify(core.get_threat_level())

@app.get("/api/settings")
def api_settings(): return jsonify(core.settings)

@app.post("/api/settings")
def api_update_settings(): return jsonify(core.update_settings(request.get_json(force=True) or {}))

@app.post("/api/folders")
def api_add_folder():
    folder = ((request.get_json(force=True) or {}).get("folder") or "").strip()
    if not folder: return jsonify({"error": "folder required"}), 400
    try: return jsonify({"folders": core.add_folder(folder)})
    except ValueError as e: return jsonify({"error": str(e)}), 400

@app.delete("/api/folders")
def api_remove_folder():
    folder = ((request.get_json(force=True) or {}).get("folder") or "").strip()
    if not folder: return jsonify({"error": "folder required"}), 400
    return jsonify({"folders": core.remove_folder(folder)})

@app.post("/api/agent/start")
def api_start():
    try:
        core.start()
        return jsonify({"status": core.status})
    except Exception as e: return jsonify({"error": str(e)}), 400

@app.post("/api/agent/stop")
def api_stop():
    core.stop()
    return jsonify({"status": core.status})

@app.post("/api/canaries/recreate")
def api_recreate_canaries(): return jsonify({"recreated": canary.recreate_missing_canaries(core.settings["protected_folders"])})

@app.get("/")
def index(): return render_template("index.html")
